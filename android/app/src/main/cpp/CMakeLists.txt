cmake_minimum_required(VERSION 3.22)
project(LumoraAndroid VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../../..)
list(APPEND CMAKE_MODULE_PATH ${REPO_ROOT}/cmake)
include(LumoraWarnings)
add_subdirectory(${REPO_ROOT}/core ${CMAKE_BINARY_DIR}/core)
add_library(lumora_jni SHARED lumora_jni.cpp)
target_link_libraries(lumora_jni PRIVATE lumora_core)
